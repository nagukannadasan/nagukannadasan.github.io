{{- /* Input: a string (URL, text or bare 11-char ID). Returns the YouTube video ID or "". */ -}} {{- $s := . | string -}} {{- $id := "" -}} {{- $m := findRESubmatch `(?:youtube\.com/(?:watch\?(?:[^\s&#]*&)?v=|embed/|shorts/|live/)|youtu\.be/)([A-Za-z0-9_-]{11})` $s 1 -}} {{- if $m -}} {{- $id = index (index $m 0) 1 -}} {{- else if findRE `^[A-Za-z0-9_-]{11}$` (trim $s " ") 1 -}} {{- $id = trim $s " " -}} {{- end -}} {{- return $id -}}
